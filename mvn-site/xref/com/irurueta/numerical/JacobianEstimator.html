16  package com.irurueta.numerical;
17  
18  import com.irurueta.algebra.Matrix;
19  import com.irurueta.algebra.WrongSizeException;
20  
21  /**
22   * Class to estimate the Jacobian of a multi variate and multidimensional
23   * function.
24   * This class evaluates a function at very close locations of a given input
25   * point in order to determine the Jacobian at such point.
26   * <p>
27   * For each input parameter {@code x}, the finite difference step is computed as
28   * {@code h = max(relativeStep * |x|, minStep)}. If the resulting step is exactly zero
29   * (e.g. when {@code x == 0.0} and {@code minStep == 0.0}), {@code relativeStep} is used as
30   * step. The step is finally adjusted to the value that is exactly representable
31   * as the difference between the perturbed and the original parameter.
32   * <p>
33   * By default {@code relativeStep} is {@link #EPS} and {@code minStep} is {@code 0.0},
34   * which preserves the behaviour of a purely relative step.
35   * Setting an absolute minimum step assumes that parameters are scaled to order 1,
36   * since the same absolute step is used regardless of the magnitude of the parameter's
37   * natural scale.
38   * <p>
39   * Instances of this class reuse internal work arrays between calls to avoid
40   * allocations, hence they are not thread-safe nor re-entrant: a single instance
41   * must not be used concurrently from several threads, nor from within the
42   * evaluated function itself.
43   */
44  public class JacobianEstimator {
45  
46      /**
47       * Constant considered as machine precision.
48       * This is the default relative step used to estimate the Jacobian.
49       */
50      public static final double EPS = 1e-8;
51  
52      /**
53       * Default minimum absolute step.
54       */
55      private static final double DEFAULT_MIN_STEP = 0.0;
56  
57      /**
58       * Listener to evaluate a multivariate function.
59       */
60      private final MultiVariateFunctionEvaluatorListener listener;
61  
62      /**
63       * Relative step respect to the magnitude of each parameter.
64       */
65      private double relativeStep = EPS;
66  
67      /**
68       * Minimum absolute step.
69       */
70      private double minStep = DEFAULT_MIN_STEP;
71  
72      /**
73       * Internal array to hold input parameter's values.
74       */
75      private double[] xh;
76  
77      /**
78       * Internal array to hold function values at the evaluated point.
79       */
80      private double[] fold;
81  
82      /**
83       * Internal array to hold function values at the perturbed point.
84       */
85      private double[] fh;
86  
87      /**
88       * Constructor.
89       * Uses {@link #EPS} as relative step and {@code 0.0} as minimum step.
90       *
91       * @param listener listener to evaluate a multivariate function.
92       */
93      public JacobianEstimator(final MultiVariateFunctionEvaluatorListener listener) {
94          this(listener, EPS, DEFAULT_MIN_STEP);
95      }
96  
97      /**
98       * Constructor.
99       *
100      * @param listener     listener to evaluate a multivariate function.
101      * @param relativeStep relative step respect to the magnitude of each parameter. Must be
102      *                     strictly positive.
103      * @param minStep      minimum absolute step. Must be positive or zero.
104      * @throws IllegalArgumentException if {@code relativeStep} is not strictly positive (or is
105      *                                  NaN or infinite), or {@code minStep} is negative (or is NaN or infinite).
106      */
107     public JacobianEstimator(
108             final MultiVariateFunctionEvaluatorListener listener, final double relativeStep, final double minStep) {
109         this.listener = listener;
110         this.relativeStep = checkRelativeStep(relativeStep);
111         this.minStep = checkMinStep(minStep);
112     }
113 
114     /**
115      * Gets relative step respect to the magnitude of each parameter.
116      * Default value is {@link #EPS}.
117      *
118      * @return relative step.
119      */
120     public double getRelativeStep() {
121         return relativeStep;
122     }
123 
124     /**
125      * Sets relative step respect to the magnitude of each parameter.
126      *
127      * @param relativeStep relative step. Must be strictly positive.
128      * @throws IllegalArgumentException if provided value is not strictly positive, or is NaN or infinite.
129      */
130     public void setRelativeStep(final double relativeStep) {
131         this.relativeStep = checkRelativeStep(relativeStep);
132     }
133 
134     /**
135      * Gets minimum absolute step.
136      * Default value is {@code 0.0}.
137      * An absolute minimum step assumes that parameters are scaled to order 1.
138      *
139      * @return minimum absolute step.
140      */
141     public double getMinStep() {
142         return minStep;
143     }
144 
145     /**
146      * Sets minimum absolute step.
147      * An absolute minimum step assumes that parameters are scaled to order 1.
148      *
149      * @param minStep minimum absolute step. Must be positive or zero.
150      * @throws IllegalArgumentException if provided value is negative, NaN or infinite.
151      */
152     public void setMinStep(final double minStep) {
153         this.minStep = checkMinStep(minStep);
154     }
155 
156     /**
157      * Returns the Jacobian of a multivariate function at provided point.
158      *
159      * @param point input point.
160      * @return jacobian.
161      * @throws EvaluationException raised if function cannot be evaluated.
162      */
163     public Matrix jacobian(final double[] point) throws EvaluationException {
164         try {
165             final var result = new Matrix(listener.getNumberOfVariables(), point.length);
166             jacobian(point, result);
167             return result;
168         } catch (final WrongSizeException e) {
169             throw new EvaluationException(e);
170         }
171     }
172 
173     /**
174      * Sets estimated jacobian in provided result matrix of a multivariate
175      * function at provided point.
176      * This method is preferred respect to jacobian(double[]) because result
177      * matrix can be reused and hence is more memory efficient.
178      * The step used for each parameter {@code x} is
179      * {@code max(relativeStep * |x|, minStep)}, or {@code relativeStep} if such value is zero.
180      *
181      * @param point  input point.
182      * @param result output matrix containing estimated jacobian. This parameter
183      *               must be a matrix having the same number of columns as the length of the
184      *               point and the same number of rows as the number of variables returned by
185      *               function evaluations.
186      * @throws EvaluationException      raised if function cannot be evaluated.
187      * @throws IllegalArgumentException if size of result is not valid.
188      */
189     public void jacobian(final double[] point, final Matrix result) throws EvaluationException {
190         final var numdims = point.length;
191         final var numvars = listener.getNumberOfVariables();
192         if (result.getColumns() != numdims) {
193             throw new IllegalArgumentException();
194         }
195         if (result.getRows() != numvars) {
196             throw new IllegalArgumentException();
197         }
198 
199         if (fold == null || fold.length != numvars) {
200             fold = new double[numvars];
201             fh = new double[numvars];
202         }
203         if (xh == null || xh.length != numdims) {
204             xh = new double[numdims];
205         }
206         System.arraycopy(point, 0, xh, 0, numdims);
207 
208         double temp;
209         double h;
210         listener.evaluate(point, fold);
211         for (var j = 0; j < numdims; j++) {
212             temp = point[j];
213             h = computeStep(temp);
214             xh[j] = temp + h;
215             h = xh[j] - temp;
216             listener.evaluate(xh, fh);
217             xh[j] = temp;
218             for (var i = 0; i < numvars; i++) {
219                 result.setElementAt(i, j, (fh[i] - fold[i]) / h);
220             }
221         }
222     }
223 
224     /**
225      * Computes the step used to perturb an input parameter.
226      * The step is {@code max(relativeStep * |x|, minStep)}, and if such value is zero, then
227      * {@code relativeStep} is used instead.
228      * Notice that the actually applied step might differ slightly because of finite machine precision.
229      *
230      * @param x value of the input parameter to be perturbed.
231      * @return step to be used.
232      */
233     protected double computeStep(final double x) {
234         final var h = Math.max(relativeStep * Math.abs(x), minStep);
235         // Trick to reduce finite-precision error
236         return h == 0.0 ? relativeStep : h;
237     }
238 
239     /**
240      * Validates a relative step.
241      *
242      * @param relativeStep value to be checked.
243      * @return provided value if valid.
244      * @throws IllegalArgumentException if value is not strictly positive, or is NaN or infinite.
245      */
246     private static double checkRelativeStep(final double relativeStep) {
247         if (!(relativeStep > 0.0) || Double.isInfinite(relativeStep)) {
248             throw new IllegalArgumentException("relativeStep must be strictly positive and finite");
249         }
250         return relativeStep;
251     }
252 
253     /**
254      * Validates a minimum step.
255      *
256      * @param minStep value to be checked.
257      * @return provided value if valid.
258      * @throws IllegalArgumentException if value is negative, NaN or infinite.
259      */
260     private static double checkMinStep(final double minStep) {
261         if (!(minStep >= 0.0) || Double.isInfinite(minStep)) {
262             throw new IllegalArgumentException("minStep must be positive or zero and finite");
263         }
264         return minStep;
265     }
