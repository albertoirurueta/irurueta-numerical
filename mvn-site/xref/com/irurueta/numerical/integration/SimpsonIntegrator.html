16  package com.irurueta.numerical.integration;
17  
18  import com.irurueta.numerical.EvaluationException;
19  import com.irurueta.numerical.SingleDimensionFunctionEvaluatorListener;
20  
21  /**
22   * Base integrator for implementations based on Simpson's method.
23   * Simpson's method is an optimization of Trapezoidal quadrature integrator.
24   * Implementations of this class will in general be more efficient than
25   * Trapezoidal quadrature integrators (i.e., require fewer function evaluations) when the function
26   * to be integrated has a finite fourth derivative (i.e., a continuous third derivative).
27   *
28   * <p>
29   * The extrapolation step combines the current quadrature estimate {@code st} and the previous one {@code ost} as
30   * {@code s = (ratio * st - ost) / (ratio - 1)}. The ratio is the factor by which the quadrature error is reduced
31   * on each refinement: {@code 2^2 = 4} for quadratures halving their step (e.g., trapezoidal), which yields
32   * {@code (4 * st - ost) / 3}, and {@code 3^2 = 9} for quadratures tripling their step (e.g., mid-point), which
33   * yields {@code (9 * st - ost) / 8}.
34   * See Numerical Recipes, 3rd ed., section 4.2 (p. 165) and section 4.4 (p. 169).
35   *
36   * @param <T> a quadrature.
37   */
38  public abstract class SimpsonIntegrator<T extends Quadrature> extends Integrator {
39  
40      /**
41       * Default accuracy.
42       */
43      public static final double EPS = 1e-10;
44  
45      /**
46       * Extrapolation ratio for quadratures that halve their step on each refinement (2^2).
47       * Leads to extrapolation {@code (4 * st - ost) / 3}.
48       */
49      protected static final double TRAPEZOIDAL_RATIO = 4.0;
50  
51      /**
52       * Extrapolation ratio for quadratures that triple their step on each refinement (3^2).
53       * Leads to extrapolation {@code (9 * st - ost) / 8}.
54       */
55      protected static final double TRIPLING_RATIO = 9.0;
56  
57      /**
58       * Default maximum number of allowed steps.
59       */
60      protected static final int DEFAULT_JMAX = 20;
61  
62      /**
63       * Maximum number of allowed steps for quadratures tripling their step on each refinement.
64       */
65      protected static final int TRIPLING_JMAX = 14;
66  
67      /**
68       * Minimum required number of steps.
69       */
70      private static final int JMIN = 5;
71  
72      /**
73       * Quadrature used for integration.
74       */
75      private final T q;
76  
77      /**
78       * Required accuracy.
79       */
80      private final double eps;
81  
82      /**
83       * Extrapolation ratio.
84       */
85      private final double ratio;
86  
87      /**
88       * Maximum number of allowed steps.
89       */
90      private final int jmax;
91  
92      /**
93       * Constructor using halving step extrapolation ratio and default maximum number of steps.
94       *
95       * @param q   Quadrature used for integration.
96       * @param eps Required accuracy.
97       */
98      protected SimpsonIntegrator(final T q, final double eps) {
99          this(q, eps, TRAPEZOIDAL_RATIO, DEFAULT_JMAX);
100     }
101 
102     /**
103      * Constructor.
104      *
105      * @param q     Quadrature used for integration.
106      * @param eps   Required accuracy.
107      * @param ratio Extrapolation ratio (4.0 for quadratures halving their step, 9.0 for tripling their step).
108      *              Must be greater than 1; otherwise the extrapolation divides by zero (or flips sign) and
109      *              integration never converges.
110      * @param jmax  Maximum number of allowed steps. Must be greater than 6, since convergence is only checked
111      *              after the minimum number of steps; otherwise integration always fails.
112      */
113     protected SimpsonIntegrator(final T q, final double eps, final double ratio, final int jmax) {
114         this.q = q;
115         this.eps = eps;
116         this.ratio = ratio;
117         this.jmax = jmax;
118     }
119 
120     /**
121      * Integrates function between provided lower and upper limits.
122      *
123      * @return result of integration.
124      * @throws IntegrationException if integration fails for numerical reasons.
125      */
126     @Override
127     public double integrate() throws IntegrationException {
128         try {
129             double s;
130             double st;
131             var ost = 0.0;
132             var os = 0.0;
133             for (var j = 0; j < jmax; j++) {
134                 st = q.next();
135                 s = (ratio * st - ost) / (ratio - 1.0);
136                 if (j > JMIN && (Math.abs(s - os) < eps * Math.abs(os) || (s == 0.0 && os == 0.0))) {
137                     return s;
138                 }
139                 os = s;
140                 ost = st;
141             }
142         } catch (final EvaluationException e) {
143             throw new IntegrationException(e);
144         }
145 
146         // Too many steps
147         throw new IntegrationException();
148     }
149 
150     /**
151      * Gets type of integrator.
152      *
153      * @return type of integrator.
154      */
155     @Override
156     public IntegratorType getIntegratorType() {
157         return IntegratorType.SIMPSON;
158     }
159 
160     /**
161      * Creates an integrator using Simpson's method.
162      *
163      * @param a              Lower limit of integration.
164      * @param b              Upper limit of integration.
165      * @param listener       listener to evaluate a single dimension function at required points.
166      * @param eps            required accuracy.
167      * @param quadratureType quadrature type.
168      * @return created integrator.
169      * @throws IllegalArgumentException if provided quadrature type is not supported.
170      */
171     public static SimpsonIntegrator<Quadrature> create(
172             final double a, final double b, final SingleDimensionFunctionEvaluatorListener listener, final double eps,
173             final QuadratureType quadratureType) {
174         return switch (quadratureType) {
175             case TRAPEZOIDAL -> cast(new SimpsonTrapezoidalQuadratureIntegrator(a, b, listener, eps));
176             case MID_POINT -> cast(new SimpsonMidPointQuadratureIntegrator(a, b, listener, eps));
177             case INFINITY_MID_POINT -> cast(new SimpsonInfinityMidPointQuadratureIntegrator(a, b, listener, eps));
178             case LOWER_SQUARE_ROOT_MID_POINT ->
179                     cast(new SimpsonLowerSquareRootMidPointQuadratureIntegrator(a, b, listener, eps));
180             case UPPER_SQUARE_ROOT_MID_POINT ->
181                     cast(new SimpsonUpperSquareRootMidPointQuadratureIntegrator(a, b, listener, eps));
182             case DOUBLE_EXPONENTIAL_RULE ->
183                     cast(new SimpsonDoubleExponentialRuleQuadratureIntegrator(a, b, listener, eps));
184             default -> throw new IllegalArgumentException();
185         };
186     }
187 
188     /**
189      * Creates an integrator using Simpson's method and having default accuracy.
190      *
191      * @param a              Lower limit of integration.
192      * @param b              Upper limit of integration.
193      * @param listener       listener to evaluate a single dimension function at required points.
194      * @param quadratureType quadrature type.
195      * @return created integrator.
196      * @throws IllegalArgumentException if provided quadrature type is not supported.
197      */
198     public static SimpsonIntegrator<Quadrature> create(
199             final double a, final double b, final SingleDimensionFunctionEvaluatorListener listener,
200             final QuadratureType quadratureType) {
201         return switch (quadratureType) {
202             case TRAPEZOIDAL -> cast(new SimpsonTrapezoidalQuadratureIntegrator(a, b, listener));
203             case MID_POINT -> cast(new SimpsonMidPointQuadratureIntegrator(a, b, listener));
204             case INFINITY_MID_POINT -> cast(new SimpsonInfinityMidPointQuadratureIntegrator(a, b, listener));
205             case LOWER_SQUARE_ROOT_MID_POINT ->
206                     cast(new SimpsonLowerSquareRootMidPointQuadratureIntegrator(a, b, listener));
207             case UPPER_SQUARE_ROOT_MID_POINT ->
208                     cast(new SimpsonUpperSquareRootMidPointQuadratureIntegrator(a, b, listener));
209             case DOUBLE_EXPONENTIAL_RULE -> cast(new SimpsonDoubleExponentialRuleQuadratureIntegrator(a, b, listener));
210             default -> throw new IllegalArgumentException();
211         };
212     }

214     /**
215      * Creates an integrator using Simpson's method and default quadrature type.
216      *
217      * @param a        Lower limit of integration.
218      * @param b        Upper limit of integration.
219      * @param listener listener to evaluate a single dimension function at required points.
220      * @param eps      required accuracy.
221      * @return created integrator.
222      */
223     public static SimpsonIntegrator<Quadrature> create(
224             final double a, final double b, final SingleDimensionFunctionEvaluatorListener listener, final double eps) {
225         return create(a, b, listener, eps, DEFAULT_QUADRATURE_TYPE);
226     }

228     /**
229      * Creates an integrator using Simpson's method and having default accuracy and quadrature type.
230      *
231      * @param a        Lower limit of integration.
232      * @param b        Upper limit of integration.
233      * @param listener listener to evaluate a single dimension function at required points.
234      * @return created integrator.
235      */
236     public static SimpsonIntegrator<Quadrature> create(
237             final double a, final double b, final SingleDimensionFunctionEvaluatorListener listener) {
238         return create(a, b, listener, DEFAULT_QUADRATURE_TYPE);
239     }

241     /**
242      * Casts integrator to a quadrature integrator without wildcard parameter.
243      *
244      * @param integrator integrator to be cast.
245      * @return cast integrator.
246      */
247     private static SimpsonIntegrator<Quadrature> cast(final SimpsonIntegrator<?> integrator) {
248         //noinspection unchecked
249         return (SimpsonIntegrator<Quadrature>) integrator;
250     }
