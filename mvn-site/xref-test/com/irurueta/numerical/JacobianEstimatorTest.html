135     @Test
136     void testJacobian() throws EvaluationException, WrongSizeException {
137 
138         for (var t = 0; t < TIMES; t++) {
139             final var randomizer = new UniformRandomizer();
140 
141             ndims = randomizer.nextInt(MIN_DIMS, MAX_DIMS);
142             nvars = randomizer.nextInt(MIN_VARS, MAX_VARS);
143 
144             minimums = Matrix.createWithUniformRandomValues(nvars, ndims, MIN_EVAL_POINT, MAX_EVAL_POINT);
145             final var point = new double[ndims];
146             randomizer.fill(point, MIN_EVAL_POINT, MAX_EVAL_POINT);
147             offsets = new double[nvars];
148             randomizer.fill(offsets, MIN_OFFSET, MAX_OFFSET);
149 
150             widths = Matrix.createWithUniformRandomValues(nvars, ndims, MIN_WIDTH, MAX_WIDTH);
151 
152             final var estimator = new JacobianEstimator(this);
153 
154             final var jacobian1 = estimator.jacobian(point);
155             final var jacobian2 = new Matrix(nvars, ndims);
156             estimator.jacobian(point, jacobian2);
157 
158             // check correctness
159             final var jacobian3 = jacobian(point);
160             assertNotNull(jacobian3);
161             for (var j = 0; j < nvars; j++) {
162                 for (var i = 0; i < ndims; i++) {
163                     assertEquals(jacobian1.getElementAt(j, i), jacobian3.getElementAt(j, i), ABSOLUTE_ERROR);
164                     assertEquals(jacobian2.getElementAt(j, i), jacobian3.getElementAt(j, i), ABSOLUTE_ERROR);
165                 }
166             }
167         }
168     }

304      * Checks that using the same relative and minimum step yields a step of {@code step * max(1, |x|)}.
305      */
306     @Test
307     void jacobian_whenRelativeAndMinStepAreEqual_usesStepTimesMaxOfOneAndAbsoluteValue() throws EvaluationException {
308         setUpFunction(RECORDING_DIMS, 1);
309         final var point = new double[]{0.5, -3.0, 20.0};
310         final var estimator = new JacobianEstimator(this, CUSTOM_STEP, CUSTOM_STEP);
311 
312         recordedPoints = new ArrayList<>();
313         estimator.jacobian(point);
314 
315         assertEquals(RECORDING_DIMS + 1, recordedPoints.size());
316         for (var j = 0; j < RECORDING_DIMS; j++) {
317             final var step = recordedPoints.get(j + 1)[j] - point[j];
318             final var expected = CUSTOM_STEP * Math.max(1.0, Math.abs(point[j]));
319             assertEquals(expected, step, STEP_TOLERANCE);
320         }

323     /**
324      * Checks that for small parameters a minimum step is closer to the analytical Jacobian than defaults.
325      */
326     @Test
327     void jacobian_whenSmallParametersAndMinStep_isMoreAccurateThanDefaults() throws EvaluationException {
328         linearMode = true;
329         nvars = 1;
330         final var defaults = new JacobianEstimator(this);
331         final var withMinStep = new JacobianEstimator(this, CUSTOM_STEP, CUSTOM_STEP);
332 
333         for (final var value : SMALL_PARAMETERS) {
334             final var point = new double[]{value};
335 
336             final var defaultsError = Math.abs(defaults.jacobian(point).getElementAt(0, 0) - LINEAR_SLOPE);
337             final var minStepError = Math.abs(withMinStep.jacobian(point).getElementAt(0, 0) - LINEAR_SLOPE);
338 
339             assertTrue(minStepError < defaultsError);
340             assertTrue(minStepError < SMALL_PARAMETER_ERROR);
341         }
342     }

359     @Override
360     public void evaluate(final double[] point, final double[] result) {
361         if (recordedPoints != null) {
362             recordedPoints.add(point.clone());
363         }
364         if (linearMode) {
365             result[0] = LINEAR_OFFSET + LINEAR_SLOPE * point[0];
366             return;
367         }
368 
369         final var dims = Math.min(Math.min(point.length, minimums.getColumns()), widths.getColumns());
370         final var vars = result.length;
371 
372         for (var j = 0; j < vars; j++) {
373             var value = 1.0;
374             for (var i = 0; i < dims; i++) {
375                 value *= Math.pow(point[i] - minimums.getElementAt(j, i), 2.0) / widths.getElementAt(j, i);
376             }
377 
378             value += offsets[j];
379 
380             result[j] = value;
381         }
382     }

389     /**
390      * Sets up the quadratic test function with deterministic parameters.
391      *
392      * @param dims number of input dimensions.
393      * @param vars number of function variables.
394      */
395     private void setUpFunction(final int dims, final int vars) {
396         ndims = dims;
397         nvars = vars;
398         linearMode = false;
399         recordedPoints = null;
400         try {
401             minimums = new Matrix(vars, dims);
402             widths = Matrix.createWithUniformRandomValues(vars, dims, MIN_WIDTH, MAX_WIDTH);
403         } catch (final WrongSizeException e) {
404             throw new IllegalStateException(e);
405         }
406         offsets = new double[vars];
407     }

409     /**
410      * Estimates the Jacobian using the original step rule: {@code EPS * |x|}, or {@code EPS} if zero.
411      *
412      * @param point input point.
413      * @return estimated Jacobian.
414      * @throws WrongSizeException if matrix cannot be created.
415      */
416     private Matrix originalRuleJacobian(final double[] point) throws WrongSizeException {
417         final var numdims = point.length;
418         final var fold = new double[nvars];
419         final var fh = new double[nvars];
420         final var xh = point.clone();
421         final var result = new Matrix(nvars, numdims);
422 
423         evaluate(point, fold);
424         for (var j = 0; j < numdims; j++) {
425             final var temp = point[j];
426             var h = DEFAULT_RELATIVE_STEP * Math.abs(temp);
427             if (h == 0.0) {
428                 h = DEFAULT_RELATIVE_STEP;
429             }
430             xh[j] = temp + h;
431             h = xh[j] - temp;
432             evaluate(xh, fh);
433             xh[j] = temp;
434             for (var i = 0; i < nvars; i++) {
435                 result.setElementAt(i, j, (fh[i] - fold[i]) / h);
436             }
437         }
438         return result;
439     }
440 
441     private Matrix jacobian(final double[] params) {
442         final var dims = Math.min(Math.min(params.length, minimums.getColumns()), widths.getColumns());
443         final var vars = nvars;
444 
445         try {
446             final var jacobian = new Matrix(vars, dims);
447             for (var k = 0; k < vars; k++) {
448                 for (var j = 0; j < dims; j++) {
449                     var value = 1.0;
450                     for (var i = 0; i < dims; i++) {
451                         if (i != j) {
452                             value *= Math.pow(params[i] - minimums.getElementAt(k, i), 2.0)
453                                     / widths.getElementAt(k, i);
454                         } else {
455                             value *= 2.0 * (params[i] - minimums.getElementAt(k, i)) / widths.getElementAt(k, i);
456                         }
457                     }
458 
459                     jacobian.setElementAt(k, j, value);
460                 }
461             }
462 
463             return jacobian;
464         } catch (final WrongSizeException e) {
465             return null;
466         }
467     }
