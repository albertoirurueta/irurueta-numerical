33      private static final double ABSOLUTE_ERROR = 1e-6;

47          assertEquals(3, estimator.getMeasurementNoiseCov().getColumns());
48          assertEquals(3, estimator.getSampleAverage().length);

52          assertThrows(IllegalArgumentException.class, () -> new MeasurementNoiseCovarianceEstimator(0));

55      @Test
56      void testUpdate() throws SignalProcessingException, IOException, WrongSizeException {
57  
58          final var estimator = new MeasurementNoiseCovarianceEstimator(3);
59  
60          final var f = new File(ACCELERATION_NO_MOTION);
61          final var data = AccelerationFileLoader.load(f);
62  
63          final var sample = new double[3];
64  
65          for (var i = 0; i < data.numSamples; i++) {
66              sample[0] = data.accelerationX[i];
67              sample[1] = data.accelerationY[i];
68              sample[2] = data.accelerationZ[i];
69  
70              estimator.update(sample);
71          }
72  
73          final var measurementNoiseCov = estimator.getMeasurementNoiseCov();
74          final var sampleAverage = estimator.getSampleAverage();
75  
76          assertEquals(data.numSamples, estimator.getSampleCount());
77  
78          // compute sample average
79          final var sampleAverage2 = new double[3];
80          for (var i = 0; i < data.numSamples; i++) {
81              sampleAverage2[0] += data.accelerationX[i] / data.numSamples;
82              sampleAverage2[1] += data.accelerationY[i] / data.numSamples;
83              sampleAverage2[2] += data.accelerationZ[i] / data.numSamples;
84          }
85  
86          // check correctness of average
87          assertArrayEquals(sampleAverage, sampleAverage2, ABSOLUTE_ERROR);
88  
89          // compute centered batch population covariance
90          final var samples = new double[data.numSamples][3];
91          for (var i = 0; i < data.numSamples; i++) {
92              samples[i][0] = data.accelerationX[i];
93              samples[i][1] = data.accelerationY[i];
94              samples[i][2] = data.accelerationZ[i];
95          }
96          final var cov = batchCovariance(samples, sampleAverage2);
97  
98          // compare matrices
99          assertMatrixEquals(cov, measurementNoiseCov, RELATIVE_ERROR);
100     }

102     @Test
103     void testUpdateKnownSamples() throws SignalProcessingException {
104         final var estimator = new MeasurementNoiseCovarianceEstimator(1);
105 
106         final var expectedVariances = new double[]{0.0, 1.0, 8.0 / 3.0};
107         final var expectedMeans = new double[]{0.0, 1.0, 2.0};
108         final var values = new double[]{0.0, 2.0, 4.0};
109 
110         for (var i = 0; i < values.length; i++) {
111             final var cov = estimator.update(new double[]{values[i]});
112 
113             assertEquals(i + 1, estimator.getSampleCount());
114             assertEquals(expectedVariances[i], cov.getElementAt(0, 0), TIGHT_ERROR);
115             assertEquals(expectedMeans[i], estimator.getSampleAverage()[0], TIGHT_ERROR);
116         }
117     }

119     @Test
120     void testUpdateMultivariate() throws SignalProcessingException, WrongSizeException {
121         final var estimator = new MeasurementNoiseCovarianceEstimator(3);
122 
123         final var samples = new double[][]{
124                 {1.0, -2.0, 0.5},
125                 {3.0, 4.0, -1.5},
126                 {-2.0, 0.0, 2.5},
127                 {0.5, 1.5, 3.0},
128                 {4.0, -3.5, 0.0},
129                 {2.5, 2.0, -2.0}
130         };
131 
132         for (var n = 1; n <= samples.length; n++) {
133             final var cov = estimator.update(samples[n - 1]);
134 
135             final var subset = Arrays.copyOf(samples, n);
136             final var mean = new double[3];
137             for (final var sample : subset) {
138                 for (var j = 0; j < 3; j++) {
139                     mean[j] += sample[j] / n;
140                 }
141             }
142 
143             assertArrayEquals(mean, estimator.getSampleAverage(), TIGHT_ERROR);
144             assertMatrixEquals(batchCovariance(subset, mean), cov, TIGHT_ERROR);
145 
146             // covariance must be symmetric
147             for (var i = 0; i < 3; i++) {
148                 for (var j = i + 1; j < 3; j++) {
149                     assertEquals(cov.getElementAt(i, j), cov.getElementAt(j, i), TIGHT_ERROR);
150                 }
151             }
152         }
153     }

155     @Test
156     void testUpdateWrongSampleSize() throws SignalProcessingException {
157         final var estimator = new MeasurementNoiseCovarianceEstimator(2);
158 
159         final var sample = new double[3];
160         assertThrows(IllegalArgumentException.class, () -> estimator.update(sample));
161     }

163     private static Matrix batchCovariance(final double[][] samples, final double[] mean) throws WrongSizeException {
164         final var dims = mean.length;
165         final var cov = new Matrix(dims, dims);
166         for (final var sample : samples) {
167             for (var i = 0; i < dims; i++) {
168                 for (var j = 0; j < dims; j++) {
169                     cov.setElementAt(i, j, cov.getElementAt(i, j)
170                             + (sample[i] - mean[i]) * (sample[j] - mean[j]) / samples.length);
171                 }
172             }
173         }
174         return cov;
175     }
176 
177     private static void assertMatrixEquals(final Matrix expected, final Matrix actual, final double relativeError) {
178         assertEquals(expected.getRows(), actual.getRows());
179         assertEquals(expected.getColumns(), actual.getColumns());
180         for (var i = 0; i < expected.getRows(); i++) {
181             for (var j = 0; j < expected.getColumns(); j++) {
182                 final var e = expected.getElementAt(i, j);
183                 assertEquals(e, actual.getElementAt(i, j), relativeError * Math.max(1.0, Math.abs(e)));
184             }
185         }
186     }
